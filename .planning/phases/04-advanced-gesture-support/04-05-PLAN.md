```c
/**
 * \brief Calculate squared distance between two points (avoids sqrt for performance).
 *
 * Uses squared distance instead of actual distance to avoid floating-point math.
 * Compare against squared thresholds for zoom detection.
 *
 * @param x1, y1 First point coordinates
 * @param x2, y2 Second point coordinates
 * @return Squared distance between points
 */
static int32_t calculate_squared_distance(uint16_t x1, uint16_t y1, uint16_t x2, uint16_t y2) {
    const int32_t dx = (int32_t)x2 - (int32_t)x1;
    const int32_t dy = (int32_t)y2 - (int32_t)y1;
    return dx * dx + dy * dy;
}
```

**Note:** Uses squared distance to avoid expensive sqrt() operation.
Compare against squared thresholds (e.g., threshold² = 10000² = 100,000,000).

  <name>Task 5: Implement Zoom state in state machine</name>
  <files>quantum/digitizer_mouse_fallback.c</files>
  <action>
Add Zoom state case to handle pinch-to-zoom gesture detection.

**Add Zoom state case before Finished state (around line 312):**

```c
case Zoom: {
    // Track both finger positions
    if (contacts == 2) {
        // Find the two active fingers
        int finger_idx[2] = {0};
        int idx = 0;
        for (int i = 0; i < DIGITIZER_FINGER_COUNT && idx < 2; i++) {
            if (report->fingers[i].tip) {
                if (idx == 0) {
                    zoom_finger1_x = report->fingers[i].x;
                    zoom_finger1_y = report->fingers[i].y;
                } else {
                    zoom_finger2_x = report->fingers[i].x;
                    zoom_finger2_y = report->fingers[i].y;
                }
                idx++;
            }
        }

        // Calculate current distance
        zoom_current_distance = calculate_squared_distance(zoom_finger1_x, zoom_finger1_y, zoom_finger2_x, zoom_finger2_y);

        // Check if distance changed enough to trigger zoom
        const int32_t distance_delta = zoom_current_distance - zoom_initial_distance;
        const int32_t threshold_squared = DIGITIZER_MOUSE_ZOOM_DISTANCE_THRESHOLD * DIGITIZER_MOUSE_ZOOM_DISTANCE_THRESHOLD;

        if (abs(distance_delta) > threshold_squared) {
            // Zoom gesture detected
            if (g_via_dilemma_config.pinch_to_zoom_enabled) {
                if (distance_delta > 0) {
                    // Fingers moved apart = zoom in
                    tap_code(DIGITIZER_ZOOM_IN_KC);
                } else {
                    // Fingers moved together = zoom out
                    tap_code(DIGITIZER_ZOOM_OUT_KC);
                }
            }
            state = Finished;  // Exit after triggering zoom
        }

        // Check timeout
        if (duration > DIGITIZER_MOUSE_ZOOM_TIMEOUT) {
            state = None;  // Timeout, no zoom detected
        }
    } else if (contacts == 0) {
        state = None;  // Fingers lifted, cancel zoom
    } else {
        state = MoveScroll;  // Wrong finger count, transition to scroll
    }
    break;
}
```

This implements:
- Two-finger tracking
- Distance calculation using helper function
- Zoom in/out detection based on distance change
- VIA config check (pinch_to_zoom_enabled)
- Timeout handling
- State transition logic
